local function corner(parent, radius)
	local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, radius or 10); c.Parent = parent; return c
end
local function stroke(parent, color, thickness, transparency)
	local s = Instance.new("UIStroke"); s.Color=color or Color3.new(0,0,0); s.Thickness=thickness or 1; s.Transparency=transparency or 0.5; s.Parent=parent; return s
end
local function gradient(parent, colorSeq, rotation)
	local g = Instance.new("UIGradient"); g.Color=colorSeq; g.Rotation=rotation or 90; g.Parent=parent; return g
end
local function pad(parent, l, r, t, b)
	local p = Instance.new("UIPadding")
	p.PaddingLeft=UDim.new(0,l or 0); p.PaddingRight=UDim.new(0,r or 0)
	p.PaddingTop=UDim.new(0,t or 0); p.PaddingBottom=UDim.new(0,b or 0)
	p.Parent=parent; return p
end

local function shakeCamera(duration, strength)
	local character = localPlayer.Character
	local humanoid  = character and character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return end
	local startTime = os.clock()
	local connection
	connection = game:GetService("RunService").RenderStepped:Connect(function()
		local elapsed = os.clock()-startTime
		if elapsed >= duration then humanoid.CameraOffset=Vector3.zero; connection:Disconnect(); return end
		local fade = 1-(elapsed/duration)
		humanoid.CameraOffset = Vector3.new((math.random()-0.5)*strength*fade, (math.random()-0.5)*strength*fade, 0)
	end)
end

RunService.RenderStepped:Connect(function()
	if targetHidden or not currentTarget or currentTarget==localPlayer then targetArrow.Visible=false; return end

	local myChar     = localPlayer.Character
	local targetRoot = targetChar and targetChar:FindFirstChild("HumanoidRootPart")

	local distance = math.floor((targetRoot.Position-myRoot.Position).Magnitude)
	if billboardDistance then billboardDistance.Text=distance.."m away" end

	camera = workspace.CurrentCamera
	local viewportPoint = camera:WorldToViewportPoint(targetRoot.Position)
	local viewportWidth = camera.ViewportSize.X
	local x = math.clamp(viewportPoint.X, 80, viewportWidth-80)
	targetArrow.Position=UDim2.fromOffset(x-75, 92)
	targetArrowText.Text=distance.."m"
	targetArrow.Visible=true
end)
