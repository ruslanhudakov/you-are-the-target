local function loadPlayerData(player)
	local uid = tostring(player.UserId)

	-- Load stats
	local statsOk, statsData = pcall(function()
		return statsStore:GetAsync(uid)
	end)

	local stats = {}
	if statsOk and type(statsData) == "table" then
		for k, v in pairs(DEFAULT_STATS) do
			stats[k] = (type(statsData[k]) == type(v)) and statsData[k] or v
		end
	else
		for k, v in pairs(DEFAULT_STATS) do
			stats[k] = v
		end
	end

	-- Load inventory
	local invOk, invData = pcall(function()
		return inventoryStore:GetAsync(uid)
	end)

	local inventory
	if invOk and type(invData) == "table" then
		inventory = {
			owned    = (type(invData.owned) == "table")    and invData.owned    or DEFAULT_INVENTORY.owned,
			equipped = (type(invData.equipped) == "table") and invData.equipped or DEFAULT_INVENTORY.equipped,
		}
		-- Always ensure free defaults are owned
		for k, v in pairs(DEFAULT_INVENTORY.owned) do
			if v then inventory.owned[k] = true end
		end
		-- Validate equipped slots exist in owned
		for cat, id in pairs(DEFAULT_INVENTORY.equipped) do
			if not inventory.equipped[cat] then
				inventory.equipped[cat] = id
			end
		end
	else
		inventory = {
			owned    = {},
			equipped = {},
		}
		for k, v in pairs(DEFAULT_INVENTORY.owned) do inventory.owned[k] = v end
		for k, v in pairs(DEFAULT_INVENTORY.equipped) do inventory.equipped[k] = v end
	end

	return stats, inventory
end

-- ──────────────────────────────────────────
-- DATA STORE: SAVE
-- ──────────────────────────────────────────
local function savePlayerData(player)
	local uid = tostring(player.UserId)

	-- Save stats from leaderstats
	local leaderstats = player:FindFirstChild("leaderstats")
	if leaderstats then
		local statsData = {}
		for k in pairs(DEFAULT_STATS) do
			local val = leaderstats:FindFirstChild(k)
			statsData[k] = val and val.Value or DEFAULT_STATS[k]
		end
		pcall(function()
			statsStore:SetAsync(uid, statsData)
		end)
	end

	-- Save inventory
	local inv = playerInventory[player.UserId]
	if inv then
		pcall(function()
			inventoryStore:SetAsync(uid, {
				owned    = inv.owned,
				equipped = inv.equipped,
			})
		end)
	end
end

-- ──────────────────────────────────────────
-- LEADERSTATS SETUP
-- ──────────────────────────────────────────
local function setupLeaderstats(player, stats)
	local leaderstats = player:FindFirstChild("leaderstats")
	if not leaderstats then
		leaderstats = Instance.new("Folder")
		leaderstats.Name = "leaderstats"
		leaderstats.Parent = player
	end

	for name, default in pairs(DEFAULT_STATS) do
		local existing = leaderstats:FindFirstChild(name)
		if existing then
			existing.Value = stats[name] or default
		else
			local value = Instance.new("IntValue")
			value.Name  = name
			value.Value = stats[name] or default
			value.Parent = leaderstats
		end
	end
end

-- ──────────────────────────────────────────
-- LEADERBOARD BROADCAST
-- ──────────────────────────────────────────
local function buildLeaderboard()
	local entries = {}
	for _, player in ipairs(Players:GetPlayers()) do
		local leaderstats = player:FindFirstChild("leaderstats")
		if leaderstats then
			local level  = leaderstats:FindFirstChild("Level")
			local coins  = leaderstats:FindFirstChild("Coins")
			local wins   = leaderstats:FindFirstChild("TargetWins")
			local catches = leaderstats:FindFirstChild("Catches")
			table.insert(entries, {
				name     = player.Name,
				userId   = player.UserId,
				level    = level   and level.Value   or 1,
				coins    = coins   and coins.Value   or 0,
				wins     = wins    and wins.Value    or 0,
				catches  = catches and catches.Value or 0,
			})
		end
	end
	-- Sort by wins desc, then coins desc
	table.sort(entries, function(a, b)
		if a.wins ~= b.wins then return a.wins > b.wins end
		return a.coins > b.coins
	end)
	-- Trim to top 10
	local top = {}
	for i = 1, math.min(10, #entries) do
		top[i] = entries[i]
	end
	return top
end

local function broadcastLeaderboard()
	local data = buildLeaderboard()
	for _, player in ipairs(Players:GetPlayers()) do
		LeaderboardUpdate:FireClient(player, data)
	end
end

local function canUseLobbyTransition()
	return currentPhase == "Lobby" or currentPhase == "Calm"
end

local function getLeaderValue(player, name)
	local ls = player:FindFirstChild("leaderstats")
	return ls and ls:FindFirstChild(name)
end

local function addLeaderValue(player, name, amount)
	local v = getLeaderValue(player, name)
	if v then v.Value += amount end
end

local function getXpNeeded(level)
	return CONFIG.LEVEL_BASE_XP + ((level - 1) * CONFIG.LEVEL_XP_GROWTH)
end

local function sendToast(player, title, message, kind)
	ProgressToast:FireClient(player, title, message, kind or "info")
end

local function awardCoins(player, amount, reason)
	addLeaderValue(player, "Coins", amount)
	if reason then sendToast(player, "+"..amount.." Coins", reason, "coins") end
end

local function awardXP(player, amount, reason)
	addLeaderValue(player, "XP", amount)
	local level = getLeaderValue(player, "Level")
	local xp    = getLeaderValue(player, "XP")
	if not level or not xp then return end
	while xp.Value >= getXpNeeded(level.Value) do
		xp.Value -= getXpNeeded(level.Value)
		level.Value += 1
		addLeaderValue(player, "Coins", CONFIG.LEVEL_UP_COINS)
		sendToast(player, "LEVEL "..level.Value, "+"..CONFIG.LEVEL_UP_COINS.." Coins", "level")
	end
	if reason then sendToast(player, "+"..amount.." XP", reason, "xp") end
end

local function setLeaderMax(player, name, amount)
	local v = getLeaderValue(player, name)
	if v and amount > v.Value then v.Value = amount end
end

local function getMatchPlayers()
	local result = {}
	for _, player in ipairs(Players:GetPlayers()) do
		local h = getHumanoid(player)
		if isInMatch(player) and h and h.Health > 0 then
			table.insert(result, player)
		end
	end
	return result
end

local function getJoinedPlayers()
	local result = {}
	for _, player in ipairs(Players:GetPlayers()) do
		if isInMatch(player) then table.insert(result, player) end
	end
	return result
end

	for _, player in ipairs(getJoinedPlayers()) do
		remote:FireClient(player, ...)
	end
end

local function sendMatchState(player)
	MatchStateUpdate:FireClient(player, {
		inMatch     = isInMatch(player),
		roundActive = roundActive,
		phase       = currentPhase,
		canJoin     = (not isInMatch(player)) and canUseLobbyTransition(),
		canLeave    = isInMatch(player) and canUseLobbyTransition(),
		targetName  = currentTarget and currentTarget.Name or "",
		targetUserId = currentTarget and currentTarget.UserId or 0,
	})
end

local function sendMatchStateToAll()

end

local function getQuestState(player)
	local state = questProgress[player.UserId]
	if not state then
		state = {
			collect=0, catch=0, survive=0,
			completed={ collect=false, catch=false, survive=false },
		}
		questProgress[player.UserId] = state
	end
	return state
end

local function sendQuestUpdate(player)
	local state = getQuestState(player)
	QuestUpdate:FireClient(player, {
		collect=state.collect, collectGoal=CONFIG.QUEST_COLLECT_GOAL, collectDone=state.completed.collect,
		catch=state.catch,     catchGoal=CONFIG.QUEST_CATCH_GOAL,     catchDone=state.completed.catch,
		survive=state.survive, surviveGoal=CONFIG.QUEST_SURVIVE_GOAL, surviveDone=state.completed.survive,
		rewardCoins=CONFIG.QUEST_REWARD_COINS, rewardXP=CONFIG.QUEST_REWARD_XP,
		quests={
			{ key="collect", title="Coin Runner",     body="Collect coins",        current=state.collect, goal=CONFIG.QUEST_COLLECT_GOAL, done=state.completed.collect },
			{ key="catch",   title="Target Catcher",  body="Catch the target",     current=state.catch,   goal=CONFIG.QUEST_CATCH_GOAL,   done=state.completed.catch   },
			{ key="survive", title="Survivor",         body="Survive once as target", current=state.survive, goal=CONFIG.QUEST_SURVIVE_GOAL, done=state.completed.survive },
		},
	})
end

local function completeQuest(player, key, title)
	local state = getQuestState(player)
	if state.completed[key] then return end
	state.completed[key] = true
	awardCoins(player, CONFIG.QUEST_REWARD_COINS)
	awardXP(player, CONFIG.QUEST_REWARD_XP)
	sendToast(player, "Quest complete", title.." +"..CONFIG.QUEST_REWARD_COINS.." Coins", "quest")
	sendQuestUpdate(player)
end

local function addQuestProgress(player, key, amount)
	local state = getQuestState(player)
	if state.completed[key] then return end
	state[key] += amount
	if key=="collect" and state.collect >= CONFIG.QUEST_COLLECT_GOAL then
		completeQuest(player, key, "Coin runner")
	elseif key=="catch" and state.catch >= CONFIG.QUEST_CATCH_GOAL then
		completeQuest(player, key, "Target catcher")
	elseif key=="survive" and state.survive >= CONFIG.QUEST_SURVIVE_GOAL then
		completeQuest(player, key, "Survivor")
	else
		sendQuestUpdate(player)
	end
end

local function restoreTransparency(player)
	local saved = originalTransparency[player]
	if not saved then return end
	for inst, t in pairs(saved) do
		if inst and inst.Parent then inst.Transparency = t end
	end
	originalTransparency[player] = nil
end

local function fadeCharacter(player, transparency)
	local character = player.Character
	if not character then return end
	restoreTransparency(player)
	originalTransparency[player] = {}
	for _, inst in ipairs(character:GetDescendants()) do
		if inst:IsA("BasePart") and inst.Name ~= "HumanoidRootPart" then
			originalTransparency[player][inst] = inst.Transparency
			inst.Transparency = math.max(inst.Transparency, transparency)
		elseif inst:IsA("Decal") then
			originalTransparency[player][inst] = inst.Transparency
			inst.Transparency = math.max(inst.Transparency, transparency)
		end
	end
end

end

local function textMatches(text, terms)
	local lower = string.lower(text)
	for _, term in ipairs(terms) do
		if string.find(lower, string.lower(term), 1, true) then return true end
	end
	return false
end

local function findNamedPart(roots, terms)
	for _, root in ipairs(roots) do
		if root then
			if textMatches(root.Name, terms) then
				local p = getFirstBasePart(root); if p then return p end
			end
			for _, d in ipairs(root:GetDescendants()) do
				if textMatches(d.Name, terms) then
					local p = getFirstBasePart(d); if p then return p end
				end
			end
		end
	end
	return nil
end

end

local function disconnectCoin(part)
	local c = coinConnections[part]; if c then c:Disconnect(); coinConnections[part]=nil end

local function clearCoins()
	for part in pairs(activeCoins) do
		disconnectCoin(part)

end

	for part in pairs(activeCoins) do if part and part.Parent then count+=1 end end

local spawnCoin
local function ensureCoinCount()
	if not coinsEnabled then return end
	while countActiveCoins() < CONFIG.MAX_COINS do spawnCoin() end
end

spawnCoin = function()
	if not coinsEnabled then return end
	local positions = getCoinSpawnPositions()
	if #positions == 0 then return end
	local position = positions[math.random(1,#positions)]
	local coin = Instance.new("Part")
	coin.Name="TargetHuntCoin"; coin.Shape=Enum.PartType.Ball; coin.Size=Vector3.new(1.1,1.1,1.1)
	coin.Position=position; coin.Anchored=true; coin.CanCollide=false
	coin.Material=Enum.Material.Neon; coin.Color=Color3.fromRGB(255,205,54); coin.Parent=getCoinsFolder()
	local light = Instance.new("PointLight"); light.Brightness=1.4; light.Range=7; light.Color=coin.Color; light.Parent=coin
	activeCoins[coin] = true
	coinConnections[coin] = coin.Touched:Connect(function(hit)
		local player = playerFromTouchedPart(hit)
		if not player or not activeCoins[coin] or not isInMatch(player) then return end
		disconnectCoin(coin)
		awardCoins(player, CONFIG.COIN_VALUE)
		addLeaderValue(player, "Collects", 1)
		addQuestProgress(player, "collect", 1)
		CoinCollected:FireClient(player, CONFIG.COIN_VALUE)
		sendShopUpdate(player)
		coin:Destroy()
		task.delay(CONFIG.COIN_RESPAWN_TIME, ensureCoinCount)
	end)
end

local function finishHunt(result, hunter)
	if not roundActive then return end
	roundActive = false
	targetCaught = result == "caught"

	if currentTarget then
		restoreTransparency(currentTarget)
		resetPlayerMovement(currentTarget)
	end

	if result == "caught" then
		if hunter and hunter ~= currentTarget then
			local totalCatchCoins = CONFIG.KILLER_COINS_BONUS + currentBounty
			awardCoins(hunter, totalCatchCoins, "Target caught")
			awardXP(hunter, CONFIG.KILLER_XP_BONUS, "Target caught")
			addLeaderValue(hunter, "Catches", 1)
			addQuestProgress(hunter, "catch", 1)
			sendShopUpdate(hunter)
			fireToMatch(TargetCaught, hunter, currentTarget)
			fireToMatch(RoundAnnouncement, hunter.Name.." caught "..currentTarget.Name.."!", "caught", CONFIG.POST_ROUND_TIME)
			sendStats("caught", { hunterName=hunter.Name, rewardCoins=totalCatchCoins, rewardXP=CONFIG.KILLER_XP_BONUS, bounty=currentBounty })
			local hunterInventory = getInventory(hunter)
			local killEffectId = hunterInventory.equipped["Kill Effect"] or "kill_basic"
			fireToMatch(KillEffectTrigger, killEffectId)
		elseif currentTarget then
			fireToMatch(TargetCaught, nil, currentTarget)
			fireToMatch(RoundAnnouncement, currentTarget.Name.." was caught!", "caught", CONFIG.POST_ROUND_TIME)
			sendStats("caught", { hunterName="" })
			fireToMatch(KillEffectTrigger, "kill_basic")
		end
	elseif currentTarget then
		awardCoins(currentTarget, CONFIG.TARGET_SURVIVE_BONUS + currentBounty, "Survived the hunt")
		awardXP(currentTarget, CONFIG.KILLER_XP_BONUS, "Survived the hunt")
		addLeaderValue(currentTarget, "TargetWins", 1)
		setLeaderMax(currentTarget, "BestSurvival", CONFIG.HUNT_TIME)
		addQuestProgress(currentTarget, "survive", 1)
		sendShopUpdate(currentTarget)
		fireToMatch(RoundAnnouncement, currentTarget.Name.." survived!", "survived", CONFIG.POST_ROUND_TIME)
		sendStats("survived", { survivalTime=CONFIG.HUNT_TIME, rewardCoins=CONFIG.TARGET_SURVIVE_BONUS+currentBounty, rewardXP=CONFIG.KILLER_XP_BONUS, bounty=currentBounty })
	end

	ClearTarget:FireAllClients(result)
	BountyUpdate:FireAllClients(0)
	clearTouchConnections()
	clearCharacterConnections()
	resetAllMovement()
	applyCosmeticsToAll()
	sendMatchStateToAll()
	broadcastLeaderboard()
end

local function attachTargetCatchDetection(target)
	clearTouchConnections()
	local character = target.Character
	if not character then return end
	for _, inst in ipairs(character:GetDescendants()) do
		if inst:IsA("BasePart") then
			local connection = inst.Touched:Connect(function(hit)
				if not roundActive or currentTarget ~= target or targetCaught then return end
				if os.clock() < catchEnabledAt then return end
				local hunter = playerFromTouchedPart(hit)
				if hunter and hunter ~= target and isInMatch(hunter) then
					finishHunt("caught", hunter)
				end
			end)
			table.insert(touchConnections, connection)
		end
	end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if humanoid then
		deathConnection = humanoid.Died:Connect(function()
			if roundActive and currentTarget == target then finishHunt("caught", nil) end
		end)
	end
end

local function chooseTarget()
	local players = getMatchPlayers()
	if #players == 0 then return nil end
	if #players == 1 then return players[1] end
	local candidates = {}
	for _, p in ipairs(players) do
		if p.UserId ~= lastTargetUserId then table.insert(candidates, p) end
	end
	if #candidates == 0 then candidates = players end
	return candidates[math.random(1, #candidates)]
end

local function countPlayers() return #getMatchPlayers() end

local function waitWithTimer(phase, duration)
	for secondsLeft = duration, 1, -1 do
		sendTimer(phase, secondsLeft)
		sendMatchStateToAll()
		task.wait(1)
	end
	sendTimer(phase, 0)
end

local function runIntermission()

	ensureCoinCount(); applyCosmeticsToAll()
	for _, p in ipairs(Players:GetPlayers()) do sendQuestUpdate(p); sendShopUpdate(p) end
	sendStats("intermission")
	fireToMatch(RoundAnnouncement, "Collect coins and get ready.", "intermission", 3)
	waitWithTimer("Calm", CONFIG.INTERMISSION_TIME)
end

	currentTarget=target; lastTargetUserId=target.UserId
	sendStats("target-selected")
	fireToMatch(RoundAnnouncement, target.Name.." IS THE TARGET!", "target-selected", CONFIG.ANNOUNCE_TIME)
	fireToMatch(SetTarget, target)
	waitWithTimer("Target selected", CONFIG.ANNOUNCE_TIME)
end

local function runHunt(target)
	roundId+=1; local thisRound=roundId
	currentTarget=target; targetCaught=false; roundActive=true
	currentBounty=CONFIG.BOUNTY_BASE; catchEnabledAt=os.clock()+CONFIG.CATCH_GRACE_SECONDS

	setMovement(target, CONFIG.TARGET_SPEED, CONFIG.TARGET_JUMP_POWER)
	attachTargetCatchDetection(target)
	ensureCoinCount(); sendStats("hunt-start")
	BountyUpdate:FireAllClients(currentBounty)
	applyCosmeticsToAll(); sendMatchStateToAll()

	table.insert(characterConnections, target.CharacterAdded:Connect(function()
		if currentTarget==target and roundActive and roundId==thisRound then
			task.wait(0.25)
			teleportToArena(target)
			setMovement(target, CONFIG.TARGET_SPEED, CONFIG.TARGET_JUMP_POWER)
			attachTargetCatchDetection(target)
			applyCosmetics(target)
		end
	end))

	for secondsLeft = CONFIG.HUNT_TIME, 1, -1 do
		if not roundActive or currentTarget~=target or roundId~=thisRound then break end
		sendTimer("Hunt", secondsLeft)
		task.wait(1)
		if roundActive and currentTarget==target and roundId==thisRound then
			addLeaderValue(target, "Coins", CONFIG.TARGET_COINS_PER_SECOND)
			sendShopUpdate(target)
			setLeaderMax(target, "BestSurvival", CONFIG.HUNT_TIME-secondsLeft+1)
			currentBounty = CONFIG.BOUNTY_BASE + ((CONFIG.HUNT_TIME-secondsLeft+1)*CONFIG.BOUNTY_PER_SECOND)
			BountyUpdate:FireAllClients(currentBounty)
		end
	end

	if roundActive and currentTarget==target and roundId==thisRound then
		sendTimer("Hunt",0); finishHunt("survived",nil)
	end

	task.wait(CONFIG.POST_ROUND_TIME)
	currentTarget=nil; applyCosmeticsToAll(); sendMatchStateToAll()
end

local function joinMatch(player)
	if isInMatch(player) then sendMatchState(player); return end
	if not canUseLobbyTransition() then
		sendMatchState(player); sendToast(player, "Round in progress", "Join during the break.", "info"); return
	end
	player:SetAttribute("InMatch", true)
	teleportToArena(player); resetPlayerMovement(player); applyCosmetics(player)
	sendMatchState(player); sendToast(player, "Joined hunt", "You are now in the arena.", "info")
	if currentTarget and roundActive then
		SetTarget:FireClient(player, currentTarget)
		if player ~= currentTarget then HunterBoostUpdate:FireClient(player, 0, true) end
	end
end

local function leaveMatch(player)
	if not isInMatch(player) then sendMatchState(player); return end
	if not canUseLobbyTransition() then
		sendMatchState(player); sendToast(player, "Round in progress", "Leave after the round.", "info"); return
	end
	if player==currentTarget and roundActive then finishHunt("caught",nil) end
	player:SetAttribute("InMatch", false)
	restoreTransparency(player); resetPlayerMovement(player); clearCosmetics(player)
	teleportToLobby(player); sendMatchState(player)
	ClearTarget:FireClient(player, "lobby")
	sendToast(player, "Lobby", "Shop and quests are open here.", "info")
end

		end

	playerInventory[player.UserId] = inventory
	player:SetAttribute("InMatch", player:GetAttribute("InMatch") == true)
	sendQuestUpdate(player); sendShopUpdate(player); sendMatchState(player)
end

-- ROUND LOOP
-- ──────────────────────────────────────────
task.spawn(function()
	while true do
		while countPlayers() < CONFIG.MIN_PLAYERS do
			sendTimer("Lobby", 0)
			fireToMatch(RoundAnnouncement, "Need more players to start. Invite friends!", "waiting", 2)
			sendMatchStateToAll(); task.wait(2)
		end
		runIntermission()
		local target = chooseTarget()
		if target then runAnnouncement(target); runHunt(target)
		else task.wait(2) end
	end
end)
