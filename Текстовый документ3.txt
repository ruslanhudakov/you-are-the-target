local function corner(parent, radius)
	local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, radius or 10); c.Parent = parent; return c
end
local function stroke(parent, color, thickness, transparency)
	local s = Instance.new("UIStroke"); s.Color=color or Color3.new(0,0,0); s.Thickness=thickness or 1; s.Transparency=transparency or 0.5; s.Parent=parent; return s
end
local function gradient(parent, colorSeq, rotation)
	local g = Instance.new("UIGradient"); g.Color=colorSeq; g.Rotation=rotation or 90; g.Parent=parent; return g
end

local function waitRemote(name)
	local r = remotesFolder:WaitForChild(name, 10)
	if not r then warn("[TargetHunt] Missing remote:", name) end
	return r
end

local function buildLeaderboardEntry(rank, data)
	local isLocal = data.name == localPlayer.Name

	local row = Instance.new("Frame")
	row.Name            = "LBEntry"
	row.LayoutOrder     = rank
	row.Size            = UDim2.new(1, 0, 0, 38)
	row.BackgroundColor3 = isLocal and Color3.fromRGB(30, 40, 60) or COLOR_BG_CARD
	row.BackgroundTransparency = isLocal and 0.0 or 0.2
	row.BorderSizePixel = 0
	row.Parent          = lbScroll
	corner(row, 8)

	if isLocal then
		stroke(row, COLOR_INFO, 1.5, 0.2)
	end

	-- Rank badge
	local rankBadge = Instance.new("Frame")
	rankBadge.Size          = UDim2.fromOffset(28, 28)
	rankBadge.Position      = UDim2.fromOffset(5, 5)
	rankBadge.BackgroundColor3 = RANK_COLORS[rank] or COLOR_BG_LIGHT
	rankBadge.BackgroundTransparency = rank <= 3 and 0.0 or 0.4
	rankBadge.BorderSizePixel = 0
	rankBadge.Parent        = row
	corner(rankBadge, 7)

	local rankLabel = Instance.new("TextLabel")
	rankLabel.Size              = UDim2.fromScale(1, 1)
	rankLabel.BackgroundTransparency = 1
	rankLabel.Font              = FONT_BODY_BLACK
	rankLabel.TextColor3        = rank <= 3 and Color3.fromRGB(30, 20, 0) or COLOR_TEXT_DIM
	rankLabel.TextScaled        = true
	rankLabel.Text              = rank <= 3 and ({"1","2","3"})[rank] or tostring(rank)
	rankLabel.Parent            = rankBadge

	-- Player name
	local nameLabel = Instance.new("TextLabel")
	nameLabel.Position          = UDim2.fromOffset(38, 3)
	nameLabel.Size              = UDim2.new(1, -130, 0, 18)
	nameLabel.BackgroundTransparency = 1
	nameLabel.Font              = isLocal and FONT_BODY_BLACK or FONT_BODY
	nameLabel.TextColor3        = isLocal and COLOR_INFO or COLOR_TEXT
	nameLabel.TextScaled        = true
	nameLabel.TextXAlignment    = Enum.TextXAlignment.Left
	nameLabel.Text              = isLocal and (data.name .. " ◀") or data.name
	nameLabel.Parent            = row

	-- Level badge
	local lvlBadge = Instance.new("TextLabel")
	lvlBadge.Position          = UDim2.fromOffset(38, 22)
	lvlBadge.Size              = UDim2.fromOffset(60, 13)
	lvlBadge.BackgroundColor3  = COLOR_BG_LIGHT
	lvlBadge.BackgroundTransparency = 0.3
	lvlBadge.Font              = FONT_BODY
	lvlBadge.TextColor3        = COLOR_TEXT_DIM
	lvlBadge.TextScaled        = true
	lvlBadge.TextXAlignment    = Enum.TextXAlignment.Left
	lvlBadge.Text              = "LV " .. tostring(data.level)
	lvlBadge.Parent            = row
	corner(lvlBadge, 4)

	-- Wins stat
	local winsFrame = Instance.new("Frame")
	winsFrame.AnchorPoint      = Vector2.new(1, 0.5)
	winsFrame.Position         = UDim2.new(1, -5, 0.5, 0)
	winsFrame.Size             = UDim2.fromOffset(88, 28)
	winsFrame.BackgroundColor3 = COLOR_BG_LIGHT
	winsFrame.BackgroundTransparency = 0.4
	winsFrame.BorderSizePixel  = 0
	winsFrame.Parent           = row
	corner(winsFrame, 7)

	local winsLabel = Instance.new("TextLabel")
	winsLabel.Size             = UDim2.fromScale(1, 0.5)
	winsLabel.BackgroundTransparency = 1
	winsLabel.Font             = FONT_BODY_BLACK
	winsLabel.TextColor3       = rank == 1 and COLOR_GOLD or COLOR_TEXT
	winsLabel.TextScaled       = true
	winsLabel.Text             = "★ " .. tostring(data.wins)
	winsLabel.Parent           = winsFrame

	local coinsLabel = Instance.new("TextLabel")
	coinsLabel.Position        = UDim2.fromScale(0, 0.5)
	coinsLabel.Size            = UDim2.fromScale(1, 0.5)
	coinsLabel.BackgroundTransparency = 1
	coinsLabel.Font            = FONT_BODY
	coinsLabel.TextColor3      = COLOR_TEXT_DIM
	coinsLabel.TextScaled      = true
	coinsLabel.Text            = "◉ " .. tostring(data.coins)
	coinsLabel.Parent          = winsFrame

	return row
end

local function updateLeaderboard(data)
	clearLeaderboard()

	if not data or #data == 0 then
		local empty = Instance.new("TextLabel")
		empty.Name = "LBEntry"
		empty.Size = UDim2.new(1, 0, 0, 36)
		empty.BackgroundTransparency = 1
		empty.Font = FONT_BODY
		empty.TextColor3 = COLOR_TEXT_DIM
		empty.TextScaled = true
		empty.Text = "No players yet"
		empty.Parent = lbScroll
		lbPanel.Size = UDim2.fromOffset(260, 96)
		return
	end

	for i, entry in ipairs(data) do
		buildLeaderboardEntry(i, entry)
	end

	-- Resize panel to fit entries (max 8 visible)
	local entryCount = math.min(#data, 8)
	local panelHeight = 52 + (entryCount * 41) + 10
	TweenService:Create(lbPanel, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {
		Size = UDim2.fromOffset(260, panelHeight),
	}):Play()
end

local function setTabVisuals()
	for category, data in pairs(tabButtons) do
		if category == selectedShopCategory then
			data.button.BackgroundColor3 = data.color
			local eg = data.button:FindFirstChildOfClass("UIGradient")
			if eg then eg:Destroy() end
		else
			data.button.BackgroundColor3 = COLOR_BG_LIGHT
		end
	end
end

local function clearRows(container, rowName)
	for _, child in ipairs(container:GetChildren()) do
		if child.Name == rowName then child:Destroy() end
	end
end

local function buildItemRow(item)
	local row = Instance.new("Frame")
	row.Name             = "ItemRow"
	row.Size             = UDim2.new(1, -6, 0, 86)
	row.BackgroundColor3 = COLOR_BG_CARD
	row.BorderSizePixel  = 0
	row.Parent           = shopList
	corner(row, 12)
	stroke(row, Color3.fromRGB(0,0,0), 1, 0.7)

	local swatch = Instance.new("Frame")
	swatch.Position       = UDim2.fromOffset(13, 13)
	swatch.Size           = UDim2.fromOffset(58, 58)
	swatch.BackgroundColor3 = item.color or COLOR_TEXT
	swatch.BorderSizePixel = 0
	swatch.Parent         = row
	corner(swatch, 12)
	stroke(swatch, COLOR_TEXT, 1, 0.6)

	if item.equipped then
		local checkBadge = Instance.new("TextLabel")
		checkBadge.AnchorPoint = Vector2.new(1,1)
		checkBadge.Position    = UDim2.new(1, 4, 1, 4)
		checkBadge.Size        = UDim2.fromOffset(22, 22)
		checkBadge.BackgroundColor3 = COLOR_SUCCESS
		checkBadge.Font        = FONT_BODY_BLACK
		checkBadge.TextColor3  = COLOR_TEXT
		checkBadge.TextScaled  = true
		checkBadge.Text        = "✓"
		checkBadge.Parent      = swatch
		corner(checkBadge, 11)
	end

	local nameLabel = Instance.new("TextLabel")
	nameLabel.Position        = UDim2.fromOffset(84, 12)
	nameLabel.Size            = UDim2.new(1, -240, 0, 26)
	nameLabel.BackgroundTransparency = 1
	nameLabel.Font            = FONT_BODY_BLACK
	nameLabel.TextColor3      = COLOR_TEXT
	nameLabel.TextScaled      = true
	nameLabel.TextXAlignment  = Enum.TextXAlignment.Left
	nameLabel.Text            = item.name
	nameLabel.Parent          = row

	local descLabel = Instance.new("TextLabel")
	descLabel.Position        = UDim2.fromOffset(84, 44)
	descLabel.Size            = UDim2.new(1, -240, 0, 22)
	descLabel.BackgroundTransparency = 1
	descLabel.Font            = FONT_BODY
	descLabel.TextColor3      = COLOR_TEXT_DIM
	descLabel.TextScaled      = true
	descLabel.TextXAlignment  = Enum.TextXAlignment.Left
	descLabel.Text            = item.description
	descLabel.Parent          = row

	local actionBtn = Instance.new("TextButton")
	actionBtn.AnchorPoint    = Vector2.new(1, 0.5)
	actionBtn.Position       = UDim2.new(1, -14, 0.5, 0)
	actionBtn.Size           = UDim2.fromOffset(130, 48)
	actionBtn.BackgroundColor3 = COLOR_DANGER
	actionBtn.BorderSizePixel = 0
	actionBtn.Font           = FONT_BODY_BLACK
	actionBtn.TextColor3     = COLOR_TEXT
	actionBtn.TextScaled     = true
	actionBtn.Parent         = row
	corner(actionBtn, 10)

	if item.equipped then
		actionBtn.Text = "EQUIPPED"
		actionBtn.BackgroundColor3 = Color3.fromRGB(40, 70, 50)
		actionBtn.AutoButtonColor = false
	elseif item.owned then
		actionBtn.Text = "EQUIP"
		actionBtn.BackgroundColor3 = COLOR_INFO
		actionBtn.Activated:Connect(function() RequestEquipItem:FireServer(item.id) end)
	else
		actionBtn.Text = "◉ " .. tostring(item.cost)
		actionBtn.Activated:Connect(function() RequestPurchase:FireServer(item.id) end)
	end
end

local function updateShop()
	setTabVisuals()
	clearRows(shopList, "ItemRow")

	if not latestShopPayload then coinsLabel.Text = "◉ ..."; return end
	coinsLabel.Text = "◉ " .. tostring(latestShopPayload.coins or 0)

	local anyItem = false
	for _, item in ipairs(latestShopPayload.items or {}) do
		if item.category == selectedShopCategory then
			buildItemRow(item); anyItem = true
		end
	end

	if not anyItem then
		local el = Instance.new("TextLabel"); el.Name="ItemRow"
		el.Size=UDim2.new(1,-6,0,60); el.BackgroundTransparency=1
		el.Font=FONT_BODY; el.TextColor3=COLOR_TEXT_DIM; el.TextScaled=true
		el.Text="Coming soon."; el.Parent=shopList
	end
end

local function applyHuntHudVisibility(payload)
	local huntGui = playerGui:FindFirstChild("TargetHuntGui")
	if not huntGui then return end
	local inMatch = payload and payload.inMatch == true
	local topBar = huntGui:FindFirstChild("TopBar"); if topBar then topBar.Visible = inMatch end
	local questPnl = huntGui:FindFirstChild("QuestPanel"); if questPnl then questPnl.Visible = inMatch end
	if not inMatch then
		for _, name in ipairs({"TargetArrow","TargetAbilityButton","HunterBoostButton","ResultPanel"}) do
			local el = huntGui:FindFirstChild(name); if el then el.Visible = false end
		end
	end
end

MatchStateUpdate.OnClientEvent:Connect(function(payload)
	latestMatchPayload = payload
	isInMatch          = payload.inMatch == true

	if isInMatch then
		playLabel.Text             = "LOBBY"
		playIcon.Text              = "⬅"
		playButton.BackgroundColor3 = COLOR_BG_CARD
		-- Remove gradient for lobby state
		local eg = playButton:FindFirstChildOfClass("UIGradient"); if eg then eg:Destroy() end
		stroke(playButton, COLOR_TEXT_DIM, 1.5, 0.4)
	else
		playLabel.Text             = "PLAY"
		playIcon.Text              = "▶"
		playButton.BackgroundColor3 = COLOR_DANGER
		-- Re-add gradient
		local eg = playButton:FindFirstChildOfClass("UIGradient")
		if not eg then
			gradient(playButton, ColorSequence.new(COLOR_DANGER, COLOR_DANGER_DEEP), 90)
		end
		local es = playButton:FindFirstChildOfClass("UIStroke")
		if es then es.Color=Color3.fromRGB(255,120,130); es.Transparency=0.35 end
	end

	applyHuntHudVisibility(payload)
end)
